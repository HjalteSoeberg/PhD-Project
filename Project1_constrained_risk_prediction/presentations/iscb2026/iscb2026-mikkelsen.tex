\section{A six-year prediction problem}
\label{sec:orgb5130b9}

\begin{frame}[label={sec:org759cd7a},fragile]{Predicting the absolute 6-year risk of diabetes in presence of the competing risk of death}
\begin{align*}
F_1(6\mid x)&=P(\text{diabetes by 6 years}\mid x),\\
F_2(6\mid x)&=P(\text{death without diabetes by 6 years}\mid x).
\end{align*}

\begin{center}
\includegraphics[width=1\textwidth]{risk-plane.pdf}
\end{center}
\end{frame}

\begin{frame}[label={sec:orga94343e}]{Prediction modeling strategy}
\begin{center}
Learner library $\longrightarrow$ K-fold cross-validation
$\longrightarrow$ Super learner ensemble $\longrightarrow$ External validation
\end{center}
Candidate learners include cause-specific Cox models, IPCW binomial
regression and machine learning methods suitably adapted for competing risk outcomes.

\pause
\vfill
\begin{block}{Focus of this talk}
\begin{itemize}
\item Parametric Cox regression models for both causes estimated coherently by minimizing the prediction time horizon-specific Brier-score. A learner that ensures \(F_1(6|x)+F_2(6|x)\le 1\).
\item If and when to artifically censor subjects at the prediction time horizon.
\end{itemize}
\end{block}
\end{frame}

\begin{frame}[label={sec:org9219975}]{Setting: competing risks}
Notation
\vfill
\begin{itemize}
\item \(T\): the time between time zero\footnote{Any time in the life of a person} and the date of an event
\item \(D\in\{1,2\}\): the cause of the event
\item \(X = (X^1,...,X^p)\): a p-dimensional vector of baseline covariates
\item \(C\): the time between time zero and censoring
\item \(\widetilde{T} = \min(T,C)\): observed time
\item \(\Delta = 1\{T\leq C\}\): censoring indicator
\item \(\widetilde{D} = \Delta D\): observed event
\item \(\tau\): prediction time horizon
\end{itemize}


 \vfill
We assume conditional independent censoring and that the probability of being censored at time \(\tau\) is bounded away from zero for all \(x\). 
\end{frame}


\begin{frame}[label={sec:org65ac301}]{The parametric Cox-exponential model}
For both causes \(j\in\{1,2\}\) and censoring \(j = 0\), specify Cox regression models with constant baseline hazards:
\begin{align*}
\lambda_j(t\mid x) &=\lambda_{0j}\exp(x^\top\boldsymbol{\beta}_j),\qquad \lambda_{0j}>0,\\
\Lambda_j(t|x) &= \int^t_0\lambda_j(s|x)ds = t\lambda_j(t|x)
\end{align*}
\pause
The observed cause-specific cumulative incidence functions:
\begin{align*}
\widetilde{F}_j(t|x) &= P(\widetilde{T}\leq t, \widetilde{D} = j|x)\\
&= \int_0^t \exp\{-\Lambda_1(s-|x)-\Lambda_2(s-|x)-\Lambda_0(s-|x)\}\Lambda_j(ds|x)
\end{align*}
\end{frame}


\begin{frame}[label={sec:org84e2fd3}]{Estimation of observed risks \footnote{Munch \& Gerds (2026). The joint survival super learner: A super learner for right-censored data. Statistica Neerlandica, 80(2), e70026.}}
\begin{block}{Brier-score loss function}
\[
BSE=
\frac{1}{N}\sum_{i=1}^{N}\sum_{j=0}^{2}
\{\widetilde{F}_j(\tau\mid X_i)-1(\widetilde{D}_i= j,\widetilde{T}_i\leq \tau)\}^{2}.
\]
\end{block}
\pause

\begin{itemize}
\item Estimating the parameters of the three Cox-exponential models by minimizing the overall Brier score yields \(\hat\Lambda_1,\hat\Lambda_2, \hat\Lambda_0\)
\end{itemize}

\begin{itemize}
\item No need for IPC weights
\item Derive estimates of the absolute risks of both events \(\hat F_1, \hat F_2\) based on \(\hat\Lambda_1,\hat\Lambda_2\)
\end{itemize}
\end{frame}

\section{From framework to learner}
\label{sec:org7f3a146}

\begin{frame}[label={sec:orgd1735ed}]{Brier score estimation}
Parameter vector and the estimator:
\begin{align*}
{\theta} &= (\lambda_{00},\boldsymbol{\beta}_0,\lambda_{01},\boldsymbol{\beta}_1,\lambda_{02},\boldsymbol{\beta}_2), \\
\hat{\theta}_{BSE}&=\arg\min_{\theta}\frac{1}{N}\sum_{i=1}^{N}\sum_{j=0}^{2}
%\{\int_0^t \exp\{-\sum_{k = 0}^2\Lambda_k(s-|x)\}\Lambda_j(ds|x)
\left\{\widetilde F_j(\tau|x;\theta)-1(\widetilde{D}_i= j,\widetilde{T}_i\leq \tau)\right\}^{2}.
\end{align*}

\pause \vfill
Cause specific risk predictions can then be obtained by substituting \(\hat \theta_{BSE}\) into:
\begin{align*}
F_j(t|x) &= P(T\leq t, D = j|x)\\
&= \int_0^t \exp\{-\Lambda_1(s-|x;\theta)-\Lambda_2(s-|x;\theta)\} \Lambda_j(ds|x;\theta)
\end{align*}
\end{frame}

\begin{frame}[label={sec:orgde325a9}]{Why estimate like this?}
\alert{Working hypothesis:} 

The Cox-exponential model makes sense as we only need predictions for one specific timepoint, so we don't need the baseline hazard to vary over time.
\vfill

Estimating both cause-specific hazards this way:
\begin{itemize}
\item targets calibration and discrimination directly at the prediction horizon
\item performs well in non-proportional hazards data
\item respects the parameter space, i.e., \(F_1 + F_2 \leq 1\)
\item produces a portable parametric learner for use outside protected enviroments.
\end{itemize}
\end{frame}

\begin{frame}[label={sec:orgc9049b2}]{Artificial censoring at the prediction horizon}
Artificial censoring replaces the observed outcome by
\[
\widetilde T_i^{(4)}=\min(\widetilde T_i,4),\qquad
\widetilde D_i^{(4)}=\widetilde D_i1\{\widetilde T_i\leq4\}.
\]
It removes all event-time information after the four-year horizon. This affects the estimates of Cox models fitted using likelihood methods.

\begin{columns}[T]
\begin{column}{0.47\textwidth}
\begin{block}{It can lose predictive power}
If proportional hazards is correct, later events inform the same $\boldsymbol{\beta}_j$ that
governs hazards before four years. Discarding them wastes information.
\end{block}
\end{column}
\begin{column}{0.47\textwidth}
\begin{block}{It can gain predictive power}
If effects change after four years, an untruncated Cox fit compromises across
periods and can learn the wrong effect for the four-year target.
\end{block}
\end{column}
\end{columns}
\end{frame}

\begin{itemize}
\item Estimation targets prediction horizon
\end{itemize}

\begin{itemize}
\item The criterion function is non-convex: initialization and repeated starts matter
\begin{itemize}
\item Use partial likehood and Breslow estimates from a semi-parametric Cox regression model to find start values for hazard ratios and baseline hazard
\item Consider to switch from Brier score to logarithmic loss
\end{itemize}
\item Need to extend framework to illness-death.
\end{itemize}
